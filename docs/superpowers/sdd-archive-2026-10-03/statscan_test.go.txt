// parseStatsLine 必须与 encoding/json 解析 CallRecord 后取 9 个标量字段的结果一致。
func TestParseStatsLineMatchesEncodingJSON(t *testing.T) {
	lines := []string{
		`{"ts":"2026-10-03T12:22:32.308+08:00","model":"mimo-v2.5-free","status":"ok","prompt_tokens":100,"completion_tokens":50,"cache_read_tokens":10,"cache_creation_tokens":0,"ttft_ms":850,"output_speed":151.5,"stream":true,"duration_ms":1234}`,
		`{"ts":"2026-08-13T17:40:27.744Z","model":"deepseek-v4-flash-free","status":"fail","prompt_tokens":7,"completion_tokens":0,"cache_read_tokens":0,"err_msg":"connection refused","output_speed":0}`,
		`{"ts":"2026-10-03T12:22:32.308+08:00","model":"m","status":"ok","prompt_tokens":1,"completion_tokens":2,"cache_read_tokens":3,"cache_creation_tokens":4,"ttft_ms":5,"output_speed":0.1}`,
		// 字符串内含括号 —— 朴素括号深度会在此损坏
		`{"ts":"2026-10-03T01:02:03.000+08:00","model":"m","status":"fail","prompt_tokens":9,"completion_tokens":8,"cache_read_tokens":7,"err_msg":"dial tcp [2406::1]:443: connect: connection refused [DONE]","events":[{"type":"connect_ok","detail":""}]}`,
		// 嵌套对象
		`{"ts":"2026-10-03T01:02:03.000+08:00","model":"m","status":"ok","prompt_tokens":5,"completion_tokens":5,"meta":{"a":[1,2,{"b":"]"}]},"nodes":["n1"]}`,
		`{"ts":"2026-10-03T01:02:03.000+08:00","model":"","status":"ok","prompt_tokens":0,"completion_tokens":0,"cache_read_tokens":0,"output_speed":0}`,
	}
	for i, line := range lines {
		var got slimRec
		if !parseStatsLine([]byte(line), &got) {
			t.Fatalf("第 %d 行解析失败: %s", i, line)
		}
		var full CallRecord
		if err := json.Unmarshal([]byte(line), &full); err != nil {
			t.Fatalf("第 %d 行 encoding/json 解析失败: %v", i, err)
		}
		if got.TS != full.TS || got.Model != full.Model || got.Status != full.Status {
			t.Errorf("第 %d 行字符串字段不符: %+v vs TS=%q M=%q S=%q", i, got, full.TS, full.Model, full.Status)
		}
		if got.PromptTokens != full.PromptTokens || got.CompletionTokens != full.CompletionTokens ||
			got.CacheCreation != full.CacheCreation || got.CacheRead != full.CacheRead ||
			got.TTFTMs != full.TTFTMs {
			t.Errorf("第 %d 行整数字段不符: %+v", i, got)
		}
		// float 必须逐位相同 —— 委托 strconv.ParseFloat 才有这个保证
		if math.Float64bits(got.OutputSpeed) != math.Float64bits(full.OutputSpeed) {
			t.Errorf("第 %d 行 output_speed 位模式不符: %v vs %v", i, got.OutputSpeed, full.OutputSpeed)
		}
	}
}

// 字段顺序、缺字段、空行、坏行
func TestParseStatsLineEdgeCases(t *testing.T) {
	var r slimRec
	if !parseStatsLine([]byte(`{"status":"ok","ts":"t","model":"m"}`), &r) || r.Status != "ok" || r.TS != "t" {
		t.Errorf("字段乱序应能解析: %+v", r)
	}
	if parseStatsLine([]byte(`{`), &r) {
		t.Error("截断行应解析失败")
	}
	if parseStatsLine([]byte(``), &r) {
		t.Error("空行应解析失败")
	}
	if parseStatsLine([]byte(`[1,2]`), &r) {
		t.Error("顶层数组应解析失败")
	}
}

// 补充（不在任务书原始用例内）：真正卡住「朴素括号深度」回归的用例。
//
// 任务书用例 5 把 prompt_tokens / completion_tokens 放在 meta **之前**，
// 而生产数据里字符串内的括号又恰好配对，所以变异掉 jsonSkipContainer 的
// 字符串感知后，任务书的三个测试（含 18 万行对拍）**依然全绿**——
// 约束 3 当时只有注释保护、没有可执行保护。
//
// 判别条件有二，缺一不可：
//  1. 容器内的字符串含**游离的** ] （不是 [2406::1] / [DONE] 那种配对的）；
//  2. 需要取值的字段出现在容器**之后**，否则提前截断也看不出来。
func TestParseStatsLineStringAwareContainerSkip(t *testing.T) {
	lines := []string{
		`{"ts":"2026-10-03T01:02:03.000+08:00","model":"m","status":"ok","meta":{"b":"]"},"prompt_tokens":42,"completion_tokens":7,"ttft_ms":11,"output_speed":3.5}`,
		`{"ts":"t","meta":{"b":"a]b"},"status":"ok","prompt_tokens":1}`,
		`{"ts":"t","meta":[{"b":"]"}],"prompt_tokens":2}`,
	}
	for i, line := range lines {
		var got slimRec
		if !parseStatsLine([]byte(line), &got) {
			t.Fatalf("第 %d 行应解析成功（容器内含游离 ] 时不得提前截断）: %s", i, line)
		}
		var full CallRecord
		if err := json.Unmarshal([]byte(line), &full); err != nil {
			t.Fatalf("第 %d 行 encoding/json 解析失败: %v", i, err)
		}
		if got.PromptTokens != full.PromptTokens || got.CompletionTokens != full.CompletionTokens ||
			got.TTFTMs != full.TTFTMs || got.TS != full.TS || got.Status != full.Status {
			t.Errorf("第 %d 行字段被容器截断吞掉: got=%+v want=%+v", i, got, full)
		}
	}
}

// 全量生产数据对拍（本地跑，不入库）
func TestParseStatsLineAgainstProductionLog(t *testing.T) {
	data, err := os.ReadFile("/tmp/prod_log.jsonl")
	if err != nil {
		t.Skip("生产日志副本不在，跳过")
	}
	n, mismatch := 0, 0
	start := 0
	for i := 0; i <= len(data); i++ {
		if i == len(data) || data[i] == '\n' {
			line := data[start:i]
			start = i + 1
			if len(line) == 0 {
				continue
			}
			var got slimRec
			var full CallRecord
			if json.Unmarshal(line, &full) != nil || !parseStatsLine(line, &got) {
				mismatch++
				continue
			}
			n++
			if got.TS != full.TS || got.Model != full.Model || got.Status != full.Status ||
				got.PromptTokens != full.PromptTokens ||
				got.CompletionTokens != full.CompletionTokens ||
				got.CacheCreation != full.CacheCreation ||
				got.CacheRead != full.CacheRead ||
				got.TTFTMs != full.TTFTMs ||
				math.Float64bits(got.OutputSpeed) != math.Float64bits(full.OutputSpeed) {
				mismatch++
				if mismatch <= 3 {
					t.Errorf("不匹配 %s\n  got=%+v\n  want=%+v", line, got, full)
				}
			}
			// 不要在这里改 i：for 的后置 i++ 已把 i 带到下一字节，
			// 而 start 同步设成 i+1，两者刚好对齐。多一个 i-- 会让下一轮
			// data[start:i] 退化成 data[i+1:i] 而 panic。
		}
	}
	t.Logf("对拍 %d 行，不匹配 %d", n, mismatch)
	if n < 180000 {
		t.Errorf("只对拍了 %d 行，预期 18 万+", n)
	}
	if mismatch != 0 {
		t.Errorf("生产数据对拍 %d 处不匹配，必须为 0", mismatch)
	}
}

// 数值越界（I-1）：strconv 在 ErrRange 时返回**饱和值**而不是 0，
// 所以不能把 err 直接丢掉——那会让 MaxInt64 / +Inf 流进 slimRec。
// 后果不是数字难看：下游桶聚合的 avg += (v-avg)/n 遇 +Inf 会永久污染
// 该模型的均值，且全程不报错。
//
// 对拍基准是 encoding/json：它对这些输入一律报错并把字段留零值
// （json: cannot unmarshal number ... into Go struct field），本函数不返回
// false（脏数值不中断整行，见报告），但**必须同样留零值**。
func TestParseStatsLineOutOfRangeNumbers(t *testing.T) {
	tests := []struct {
		name string
		line string
	}{
		// —— 越界：必须挡住 ——
		{"int 正溢出", `{"ts":"t","prompt_tokens":99999999999999999999}`},
		{"int 负溢出", `{"ts":"t","completion_tokens":-99999999999999999999}`},
		{"float 正溢出", `{"ts":"t","output_speed":1e999}`},
		{"float 负溢出", `{"ts":"t","output_speed":-1e999}`},
		{"int 刚好越界一位", `{"ts":"t","ttft_ms":9999999999999999999}`},
		// Inf/NaN 字面量：非法 JSON，但 strconv.ParseFloat 对它们**返回 nil err**，
		// 所以只按 err 过滤挡不住，必须另判 IsInf/IsNaN。
		{"Inf 字面量", `{"ts":"t","output_speed":Inf}`},
		{"NaN 字面量", `{"ts":"t","output_speed":NaN}`},
		// —— 语法错：留零值是既定宽容行为，本用例锁住它不被顺手改掉 ——
		{"字符串冒充数字", `{"ts":"t","prompt_tokens":"abc"}`},
		{"null", `{"ts":"t","output_speed":null}`},
		// —— 边界：encoding/json 接受，必须原样保留 ——
		// 这两条防止把修复写成「看到 MaxInt64 就丢」这种过度纠正：
		// ParseInt("9223372036854775807") 的 err 是 nil，encoding/json 也收下它。
		{"int64 上界合法值", `{"ts":"t","prompt_tokens":9223372036854775807}`},
		{"float64 有限大值", `{"ts":"t","output_speed":1e308}`},
	}
	for _, tt := range tests {
		var r slimRec
		if !parseStatsLine([]byte(tt.line), &r) {
			t.Errorf("%s: 应解析成功（脏数值不中断整行）: %s", tt.name, tt.line)
			continue
		}
		var full CallRecord
		jerr := json.Unmarshal([]byte(tt.line), &full)
		if jerr == nil {
			// encoding/json 没报错 => 该值合法，扫描器必须逐位相同
			if r.PromptTokens != full.PromptTokens || r.CompletionTokens != full.CompletionTokens ||
				r.TTFTMs != full.TTFTMs {
				t.Errorf("%s: 合法值被丢弃 got=%+v want=%+v", tt.name, r, full)
			}
			if math.Float64bits(r.OutputSpeed) != math.Float64bits(full.OutputSpeed) {
				t.Errorf("%s: 合法浮点被丢弃 got=%v want=%v", tt.name, r.OutputSpeed, full.OutputSpeed)
			}
			continue
		}
		// encoding/json 报错 => 该值非法，整数必须留零、浮点必须有限
		if r.PromptTokens == math.MaxInt64 || r.CompletionTokens == math.MinInt64 || r.TTFTMs == math.MaxInt64 {
			t.Errorf("%s: strconv 饱和值泄漏 got=%+v（encoding/json 报错 %v）", tt.name, r, jerr)
		}
		if math.IsInf(r.OutputSpeed, 0) || math.IsNaN(r.OutputSpeed) {
			t.Errorf("%s: 非有限浮点泄漏 got=%+v（encoding/json 报错 %v）", tt.name, r, jerr)
		}
	}
}

// 转义串在**被扫描的字段**里（I-2 变异体 C）：jsonReadString 必须真的走
// strconv.Unquote 路径。生产数据里 10,649 行的转义全在 events（走
// jsonSkipContainer 字节跳过），ts/model/status 一个转义都没有，
// 所以「删掉 Unquote 分支」这种变异现有测试完全测不出来。
func TestParseStatsLineEscapedStringValues(t *testing.T) {
	lines := []string{
		`{"ts":"a\"b","model":"m\\n","status":"o\u0041","prompt_tokens":1}`,
		`{"ts":"C:\\path\\to","model":"x\\/y","status":"\u00e9t\u00e9","prompt_tokens":2}`,
		`{"ts":"t","model":"m","status":"ok","prompt_tokens":3,"ts2":1}`,
	}
	for i, line := range lines {
		var got slimRec
		if !parseStatsLine([]byte(line), &got) {
			t.Fatalf("第 %d 行应解析成功: %s", i, line)
		}
		var full CallRecord
		if err := json.Unmarshal([]byte(line), &full); err != nil {
			t.Fatalf("第 %d 行 encoding/json 解析失败: %v", i, err)
		}
		if got.TS != full.TS {
			t.Errorf("第 %d 行 TS 未解转义: got=%q want=%q", i, got.TS, full.TS)
		}
		if got.Model != full.Model {
			t.Errorf("第 %d 行 Model 未解转义: got=%q want=%q", i, got.Model, full.Model)
		}
		if got.Status != full.Status {
			t.Errorf("第 %d 行 Status 未解转义: got=%q want=%q", i, got.Status, full.Status)
		}
	}
}

// 转义串在**被跳过的容器**里（I-2 变异体 B）：jsonSkipContainer 必须跟踪
// escaped 状态，否则 `\"` 会提前关闭字符串，把字符串内容里的括号当成结构符号，
// 容器提前截断，后面需要取值的字段整条丢失。
//
// 生产数据里的 73,282 个 `\"` 恰好都不含括号（已实测：同字符串内 `\"` 之间
// 含 {}[] 的行数为 0），所以变异体 B 在 18 万行对拍上全绿——纯属侥幸。
// 下面第 1、2 条就是会打破这份侥幸的真实形状：错误信息里回显 JSON 响应体。
func TestParseStatsLineEscapedQuoteInContainerSkip(t *testing.T) {
	lines := []string{
		`{"ts":"t","meta":{"err":"parse body: {\"error\":\"rate limit\"}"},"prompt_tokens":42}`,
		`{"ts":"t","events":[{"type":"connect_error","detail":"Post \"a[b]\": boom"}],"prompt_tokens":7,"ttft_ms":3}`,
		`{"ts":"t","meta":{"a":[1,2,{"b":"]\"["}]},"prompt_tokens":5}`,
		`{"ts":"t","meta":{"b":"c\\"},"prompt_tokens":8}`,
	}
	for i, line := range lines {
		var got slimRec
		if !parseStatsLine([]byte(line), &got) {
			t.Fatalf("第 %d 行应解析成功（容器内转义引号不得提前关闭字符串）: %s", i, line)
		}
		var full CallRecord
		if err := json.Unmarshal([]byte(line), &full); err != nil {
			t.Fatalf("第 %d 行 encoding/json 解析失败: %v", i, err)
		}
		if got.PromptTokens != full.PromptTokens || got.TTFTMs != full.TTFTMs || got.TS != full.TS {
			t.Errorf("第 %d 行字段被容器截断吞掉: got=%+v want=%+v", i, got, full)
		}
	}
}
